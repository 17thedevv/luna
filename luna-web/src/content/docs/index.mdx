```luna
import <core>;

export struct Point {
    export x: f64,
    export y: f64,
};

trait Computable {
    fn magnitude_squared(self: &Self) -> f64;
}

impl Computable for Point {
    fn magnitude_squared(self: &Self) -> f64 {
        return self.x * self.x + self.y * self.y;
    }
}

fn main() -> i32 {
    dec pt = Point { x: 3.0, y: 4.0 };
    dec dist_sq = pt.magnitude_squared();
    
    // Explicit compile-time immutability
    dec total = dist_sq;
    std::io::println("Magnitude squared: {}", total);
    return 0;
}
```

        </p>
      </div>

  {/* ── 3. Editorial Deep Dive: Ownership & Borrowing ── */}
  <section class="luna-section">
    <div class="luna-editorial-grid">
      <div class="luna-editorial-text">
        <div class="luna-section-tag">Memory Management</div>
        <h3>Deterministic resource lifetimes.<br />No hidden runtime penalties.</h3>
        <p>
          In Luna, every value has a unique owning binding. When an owner goes out of scope,
          its resources are immediately reclaimed. Values can be transferred via move semantics
          or borrowed through explicit shared (`&T`) and mutable (`&rw T`) references.
        </p>
        <p>
          The compiler enforces aliasing XOR mutability at compile time, preventing mutable
          aliasing bugs before machine code is produced.
        </p>
        <ul class="luna-key-points">
          <li><strong>Zero Runtime Cost:</strong> Verification occurs entirely at compile time.</li>
          <li><strong>Explicit Mutability:</strong> Bindings are immutable by default (`dec`), made rewritable via `rw`.</li>
          <li><strong>Lifetime Contracts:</strong> Structs declare explicit lifetime invariants with `requires life(f) >= life(self)`.</li>
        </ul>
      </div>

```luna
import <alloc>;

fn process(buf: std::Vec<u8>) {
    // Takes ownership of buf; deallocated on exit
}

fn main() -> i32 {
    dec rw v = std::Vec<u8>::new();
    v.push(42);

    // Shared borrow scope: multiple readers permitted
    {
        dec r1: &std::Vec<u8> = &v;
        dec r2: &std::Vec<u8> = &v;
        std::io::println("Length: {}", r1.len());
    } // Shared borrows end here before moving v

    // Transfer ownership (move)
    process(v);

    // Error at compile-time:
    // v has been moved, cannot access
    // v.push(99);

    return 0;
}
```

```luna
// Standard C Library Interop
extern fn malloc(size: usize) -> *rw u8;
extern fn free(ptr: *rw u8);
extern fn memcpy(dst: *rw u8, src: *u8, n: usize) -> *rw u8;

export fn allocate_bytes(bytes: usize) -> *rw u8 {
    // Raw pointers are safe to declare and pass
    dec raw_ptr: *rw u8 = malloc(bytes);

    // Dereferencing raw memory requires an explicit unsafe boundary
    unsafe {
        if raw_ptr != null {
            *raw_ptr = 0;
        }
    }

    return raw_ptr;
}
```

      <div class="luna-editorial-text">
        <div class="luna-section-tag">Low-Level Interoperability</div>
        <h3>Direct hardware control.<br />Explicit safety boundaries.</h3>
        <p>
          Systems programming demands the ability to speak directly with operating systems,
          existing C libraries, and custom hardware interfaces.
        </p>
        <p>
          Luna provides clean, unencumbered FFI declarations, first-class raw pointers
          (`*T` and `*rw T`), and deliberate `unsafe` blocks. Low-level code remains isolated,
          auditable, and explicit.
        </p>
        <ul class="luna-key-points">
          <li><strong>Direct C ABI:</strong> Direct calling convention mapping without runtime bridging layers.</li>
          <li><strong>First-Class Pointers:</strong> Distinct immutable (`*T`) and mutable (`*rw T`) raw pointers.</li>
          <li><strong>Auditable Unsafe:</strong> Unsafe operations are syntactically quarantined.</li>
        </ul>

```luna
    fn len(self: &Self) -> usize;

struct Stack<T> {
    items: std::Vec<T>,
};

impl<T> Container<T> for Stack<T> {
    fn len(self: &Self) -> usize {
        return self.items.len();
    }

    fn is_empty(self: &Self) -> bool {
        return self.items.len() == 0;

}
```
