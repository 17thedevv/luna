## Ví dụ Nhanh

```luna
struct Counter {
    value: i32,
};

impl Counter {
    fn new() -> Counter {
        return Counter { value: 0 };
    }

    fn increment(self: &rw Counter) {
        self.value = self.value + 1;
    }

    fn get(self: &Counter) -> i32 {
        return self.value;
    }
}

fn main() -> i32 {
    dec rw counter = Counter::new();
    counter.increment();
    counter.increment();
    std::io::println("Đếm: {}", counter.get());
    return 0;
}
```
