// Tương tác Thư viện C Tiêu chuẩn
extern fn malloc(size: usize) -> *rw u8;
extern fn free(ptr: *rw u8);
extern fn memcpy(dst: *rw u8, src: *u8, n: usize) -> *rw u8;

export fn allocate_bytes(bytes: usize) -> *rw u8 {
    // Khai báo và truyền con trỏ thô hoàn toàn an toàn
    dec raw_ptr: *rw u8 = malloc(bytes);

    // Thao tác giải mã vùng nhớ thô bắt buộc phải đặt trong khối unsafe
    unsafe {
        if raw_ptr != null {
            *raw_ptr = 0;
        }
    }

    return raw_ptr;
}
